cb_ydt_scarlet_invasion = {
    is_triggered_only = no
    prerequisites = {
        FROM = {
            NOT = {
                AND = {
                    is_subject = yes
                    NOT = {
                        is_subject_of_type = tributary_state
                    }
                }
            }
            OR = {
                has_country_flag = ydt_scarlet_invasion_target_@ROOT
                AND = {
                    has_country_flag = ydt_anathema_invasion_target_@ROOT
                    NOT = {
                        had_country_flag = {
                            flag = ydt_anathema_invasion_target_@ROOT
                            days = 3650
                        }
                    }
                }
            }
        }
    }
    war_goal = ydt_scarlet_invasion_wargoal
}
