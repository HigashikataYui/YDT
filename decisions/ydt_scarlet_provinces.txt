    ydt_establish_balkan_province = {
        major = yes
        potential = {
            tag = WAL
            has_country_flag = ydt_unlocked_scarlet_provinces
            mission_completed = ydt_march_into_the_balkans
            NOT = {
                has_country_flag = ydt_established_balkan_province
            }
            NOT = {
                exists = YBP
            }
        }
		provinces_to_highlight = {
			OR = {
				area = slavonia_area
				area = croatia_area
				area = east_adriatic_coast_area
				area = bosnia_area
				area = serbia_area
				area = rascia_area
			}
			NOT = { owned_by = ROOT }
		}
        allow = {
            is_subject = no
            is_at_war = no
            slavonia_area = {
                type = all
                owned_by = ROOT
            }
            croatia_area = {
                type = all
                owned_by = ROOT
            }
            east_adriatic_coast_area = {
                type = all
                owned_by = ROOT
            }
            bosnia_area = {
                type = all
                owned_by = ROOT
            }
            serbia_area = {
                type = all
                owned_by = ROOT
            }
            rascia_area = {
                type = all
                owned_by = ROOT
            }
        }
        effect = {
            custom_tooltip = ydt_establish_balkan_province_tt
            hidden_effect = {
                # Add cores before release.
                every_owned_province = {
                    limit = {
                        OR = {
                            area = slavonia_area
                            area = croatia_area
                            area = east_adriatic_coast_area
                            area = bosnia_area
                            area = serbia_area
                            area = rascia_area
                        }
                    }
                    add_core = YBP
                }
                release = YBP
                every_owned_province = {
                    limit = {
                        OR = {
                            area = slavonia_area
                            area = croatia_area
                            area = east_adriatic_coast_area
                            area = bosnia_area
                            area = serbia_area
                            area = rascia_area
                        }
                    }
                    cede_province = YBP
                }
                create_subject = {
                    subject_type = ydt_scarlet_province
                    subject = YBP
                }
                YBP = {
                    change_government = monarchy
                    add_government_reform = ydt_scarlet_province_reform
                    change_religion = ROOT

                    set_capital = 4239

                    every_owned_province = {
                        add_core = YBP
                    }
                }
                set_country_flag = ydt_established_balkan_province
            }
        }
    }
