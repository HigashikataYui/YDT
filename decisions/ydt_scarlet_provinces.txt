            }

				value = 10

					area = slavonia_area

					area = bosnia_area
					area = serbia_area
					area = rascia_area
				}
			}
			4239 = {
				owned_by = ROOT
				controlled_by = ROOT
			}
		}
        effect = {
            custom_tooltip = ydt_establish_balkan_province_tt
            hidden_effect = {
                every_province = {
                    limit = {
                        OR = {
                            area = slavonia_area
                            area = croatia_area
                            area = east_adriatic_coast_area
                            area = bosnia_area
                            area = serbia_area
                            area = rascia_area
                        }
                    }
                    add_core = YBP
                }
                release = YBP
                every_owned_province = {
                    limit = {
                        OR = {
                            area = slavonia_area
                            area = croatia_area
                            area = east_adriatic_coast_area
                            area = bosnia_area
                            area = serbia_area
                            area = rascia_area
                        }
                    }
                    cede_province = YBP
                }
                create_subject = {
                    subject_type = ydt_scarlet_province
                    subject = YBP
                }
                YBP = {
                    change_government = monarchy
                    add_government_reform = ydt_scarlet_province_reform
                    change_religion = ROOT
                    set_capital = 4239
					capital_scope = {
						change_religion = animism
						add_reform_center = animism
					}
                    every_owned_province = {
						add_core = YBP
						remove_core = ROOT
					}
                }
                set_country_flag = ydt_established_balkan_province
            }
        }
    }
